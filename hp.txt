-- 拖拽功能
local dragging = false
local dragInput, dragStart, startPos

local function updateDrag(input)
    local delta = input.Position - dragStart
    mainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end

titleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = mainFrame.Position
        
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

titleBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement and dragging then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        updateDrag(input)
    end
end)

-- 创建滑块控件函数
local function createSlider(name, minValue, maxValue, defaultValue, callback)
    local container = Instance.new("Frame")
    container.Name = name .. "Container"
    container.Size = UDim2.new(1, 0, 0, 70)
    container.BackgroundTransparency = 1
    
    local label = Instance.new("TextLabel")
    label.Name = name .. "Label"
    label.Size = UDim2.new(1, 0, 0, 25)
    label.BackgroundTransparency = 1
    label.Text = name
    label.TextColor3 = Color3.fromRGB(220, 220, 240)
    label.TextSize = 16
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Font = Enum.Font.GothamSemibold
    
    local valueLabel = Instance.new("TextLabel")
    valueLabel.Name = name .. "Value"
    valueLabel.Size = UDim2.new(0, 60, 0, 25)
    valueLabel.Position = UDim2.new(1, -60, 0, 0)
    valueLabel.BackgroundTransparency = 1
    valueLabel.Text = tostring(defaultValue)
    valueLabel.TextColor3 = Color3.fromRGB(180, 220, 255)
    valueLabel.TextSize = 16
    valueLabel.TextXAlignment = Enum.TextXAlignment.Right
    valueLabel.Font = Enum.Font.GothamMedium
    
    local sliderFrame = Instance.new("Frame")
    sliderFrame.Name = name .. "SliderFrame"
    sliderFrame.Size = UDim2.new(1, 0, 0, 25)
    sliderFrame.Position = UDim2.new(0, 0, 0, 30)
    sliderFrame.BackgroundColor3 = Color3.fromRGB(50, 50, 65)
    
    local sliderCorner = Instance.new("UICorner")
    sliderCorner.CornerRadius = UDim.new(0, 4)
    sliderCorner.Parent = sliderFrame
    
    local fill = Instance.new("Frame")
    fill.Name = name .. "Fill"
    fill.Size = UDim2.new((defaultValue - minValue) / (maxValue - minValue), 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(80, 160, 255)
    
    local fillCorner = Instance.new("UICorner")
    fillCorner.CornerRadius = UDim.new(0, 4)
    fillCorner.Parent = fill
    
    local sliderButton = Instance.new("TextButton")
    sliderButton.Name = name .. "Button"
    sliderButton.Size = UDim2.new(1, 0, 1, 0)
    sliderButton.BackgroundTransparency = 1
    sliderButton.Text = ""
    
    -- 滑块交互
    local draggingSlider = false
    
    local function updateSliderValue(input)
        local relativeX = (input.Position.X - sliderFrame.AbsolutePosition.X) / sliderFrame.AbsoluteSize.X
        relativeX = math.clamp(relativeX, 0, 1)
        local value = minValue + (maxValue - minValue) * relativeX
        
        -- 根据类型取整
        if name == "平滑度" then
            value = math.floor(value * 100 + 0.5) / 100
        else
            value = math.floor(value + 0.5)
        end
        
        fill.Size = UDim2.new(relativeX, 0, 1, 0)
        valueLabel.Text = tostring(value)
        
        if callback then
            callback(value)
        end
    end
    
    sliderButton.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            draggingSlider = true
            updateSliderValue(input)
        end
    end)
    
    sliderButton.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            draggingSlider = false
        end
    end)
    
    UserInputService.InputChanged:Connect(function(input)
        if draggingSlider and input.UserInputType == Enum.UserInputType.MouseMovement then
            updateSliderValue(input)
        end
    end)
    
    -- 组装
    fill.Parent = sliderFrame
    sliderButton.Parent = sliderFrame
    label.Parent = container
    valueLabel.Parent = container
    sliderFrame.Parent = container
    
    return container
end

-- 复选框函数
local function createCheckbox(name, defaultValue, callback)
    local container = Instance.new("Frame")
    container.Name = name .. "CheckContainer"
    container.Size = UDim2.new(1, 0, 0, 40)
    container.BackgroundTransparency = 1
    
    local label = Instance.new("TextLabel")
    label.Name = name .. "Label"
    label.Size = UDim2.new(1, -50, 1, 0)
    label.BackgroundTransparency = 1
    label.Text = name
    label.TextColor3 = Color3.fromRGB(220, 220, 240)
    label.TextSize = 16
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Font = Enum.Font.GothamMedium
    
    local checkbox = Instance.new("TextButton")
    checkbox.Name = name .. "Checkbox"
    checkbox.Size = UDim2.new(0, 30, 0, 30)
    checkbox.Position = UDim2.new(1, -35, 0, 5)
    checkbox.BackgroundColor3 = defaultValue and Color3.fromRGB(80, 160, 255) or Color3.fromRGB(80, 80, 100)
    checkbox.Text = defaultValue and "✓" or ""
    checkbox.TextColor3 = Color3.fromRGB(255, 255, 255)
    checkbox.TextSize = 20
    
    local checkCorner = Instance.new("UICorner")
    checkCorner.CornerRadius = UDim.new(0, 6)
    checkCorner.Parent = checkbox
    
    checkbox.MouseButton1Click:Connect(function()
        local newValue = not (checkbox.Text == "✓")
        checkbox.Text = newValue and "✓" or ""
        checkbox.BackgroundColor3 = newValue and Color3.fromRGB(80, 160, 255) or Color3.fromRGB(80, 80, 100)
        
        if callback then
            callback(newValue)
        end
    end)
    
    label.Parent = container
    checkbox.Parent = container
    
    return container
end

-- 更新UI状态的函数
local function updateUI()
    local enabled = AimAssist:IsEnabled()
    
    -- 更新状态指示器
    statusIndicator.BackgroundColor3 = enabled and Color3.fromRGB(80, 220, 120) or Color3.fromRGB(255, 80, 80)
    statusText.Text = enabled and "已启用" .. (AimAssist:GetCurrentTarget() and " (锁定中)" or " (搜索中)") or "已禁用"
    statusText.TextColor3 = enabled and Color3.fromRGB(180, 255, 180) or Color3.fromRGB(255, 120, 120)
    toggleButton.Text = enabled and "禁用自动瞄准" or "启用自动瞄准"
    toggleButton.BackgroundColor3 = enabled and Color3.fromRGB(220, 80, 80) or Color3.fromRGB(60, 120, 220)
    
    -- 更新目标信息
    local target = AimAssist:GetCurrentTarget()
    if target then
        targetInfo.Text = "当前目标: " .. target.Name .. "\n距离: " .. tostring(math.floor(AimAssist:GetTargetDistance() or 0)) .. " 单位"
        targetInfo.TextColor3 = Color3.fromRGB(180, 255, 180)
    else
        targetInfo.Text = "当前目标: 无\n" .. (enabled and "正在搜索目标..." or "系统已禁用")
        targetInfo.TextColor3 = Color3.fromRGB(200, 200, 220)
    end
end

-- 按钮事件
toggleButton.MouseButton1Click:Connect(function()
    AimAssist:Toggle()
    updateUI()
end)

-- 更新循环
RunService.RenderStepped:Connect(updateUI)

-- 初始化UI
