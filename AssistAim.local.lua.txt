-- AssistAim.local.lua
-- LocalScript：放入 StarterPlayer -> StarterPlayerScripts
-- 功能：
--  - 辅助准心（屏幕上的准星）平滑锁定指定玩家头部/中心
--  - UI：选择目标、锁定/解锁、隐藏/显示 UI、墙壁检测开关、最大距离可调、平滑系数可调、状态显示
--  - 墙壁检测：如果本地玩家与目标之间有阻挡（墙/物体），则禁止锁定或自动取消锁定
--  - 快捷键：P 切换锁定，U 隐藏/显示 UI，[/] 切换目标
-- 注意：仅影响本地客户端的 UI / 准心，不会改变服务器或其他玩家视角

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local Workspace = game:GetService("Workspace")

local LocalPlayer = Players.LocalPlayer
local Camera = Workspace.CurrentCamera

-- 设置
local settings = {
    locked = false,
    wallCheck = true,        -- 是否检查本地玩家和目标之间的直线可见性
    maxDistance = 200,       -- 最远锁定距离 (stud)
    smoothing = 12,          -- 准心移动平滑系数（越大越快）
    targetIndex = 1,
    target = nil,
    aimOffset = Vector3.new(0, 0.5, 0), -- 在目标部位上偏移（例如指向头顶上方）
    crosshairColor = Color3.fromRGB(255, 80, 80),
}

-- 工具：返回可用于瞄准的部位（优先 Head，其次 HumanoidRootPart）
local function getAimPart(player)
    if not player or not player.Character then return nil end
    local char = player.Character
    local head = char:FindFirstChild("Head")
    if head and head:IsA("BasePart") then return head end
    local root = char:FindFirstChild("HumanoidRootPart")
    if root and root:IsA("BasePart") then return root end
    -- 回退：任何 BasePart
    for _, v in pairs(char:GetChildren()) do
        if v:IsA("BasePart") then return v end
    end
    return nil
end

-- 获取其他玩家列表（不包含自己）
local function getOtherPlayers()
    local out = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            table.insert(out, p)
        end
    end
    return out
end

local function selectTargetByIndex(idx)
    local list = getOtherPlayers()
    if #list == 0 then
        settings.target = nil
        settings.targetIndex = 1
    else
        if idx < 1 then idx = #list end
        if idx > #list then idx = 1 end
        settings.targetIndex = idx
        settings.target = list[idx]
    end
end

-- 初始选择
selectTargetByIndex(1)

-- Raycast 参数（黑名单）
local function makeRaycastParams(ignoreInstances)
    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Blacklist
    rp.FilterDescendantsInstances = ignoreInstances or {}
    rp.IgnoreWater = true
    return rp
end

-- 检查本地玩家与目标之间直线是否被阻挡（true = 无阻挡）
local function isLineOfSightClear(targetPart)
    if not targetPart then return false end
    local myPart = getAimPart(LocalPlayer)
    if not myPart then return false end
    local from = myPart.Position
    local to = targetPart.Position
    local dir = to - from
    if dir.Magnitude <= 0.001 then return true end
    local ignoreList = {}
    if LocalPlayer.Character then table.insert(ignoreList, LocalPlayer.Character) end
    if settings.target and settings.target.Character then table.insert(ignoreList, settings.target.Character) end
    local rp = makeRaycastParams(ignoreList)
    local ray = Workspace:Raycast(from, dir, rp)
    return ray == nil
end

local ui = createUI()

-- UI 更新函数
local function refreshTargetLabel()
    if settings.target and settings.target.Name then
        ui.TargetLabel.Text = "目标： " .. settings.target.Name
    else
        ui.TargetLabel.Text = "目标： 无"
    end
end

-- UI 事件连接
ui.PrevBtn.MouseButton1Click:Connect(function()
    selectTargetByIndex(settings.targetIndex - 1)
    refreshTargetLabel()
end)
ui.NextBtn.MouseButton1Click:Connect(function()
    selectTargetByIndex(settings.targetIndex + 1)
    refreshTargetLabel()
end)

ui.WallToggle.MouseButton1Click:Connect(function()
    settings.wallCheck = not settings.wallCheck
    ui.WallToggle.Text = "墙壁检测: " .. (settings.wallCheck and "开" or "关")
end)

ui.HideBtn.MouseButton1Click:Connect(function()
    ui.ScreenGui.Enabled = false
end)

ui.DistBox.FocusLost:Connect(function(enterPressed)
    local v = tonumber(ui.DistBox.Text)
    if v and v > 0 then
        settings.maxDistance = v
    else
        ui.DistBox.Text = tostring(settings.maxDistance)
    end
end)

ui.SmoothBox.FocusLost:Connect(function(enterPressed)
    local v = tonumber(ui.SmoothBox.Text)
    if v and v > 0 then
        settings.smoothing = v
    else
        ui.SmoothBox.Text = tostring(settings.smoothing)
    end
end)

-- 切换锁定（尝试锁定前会验证距离与墙壁）
local function tryToggleLock()
    if not settings.target then
        ui.StatusLabel.Text = "状态: 请选择目标"
        return
    end
    if not settings.locked then
        local aimPart = getAimPart(settings.target)
        if not aimPart then
            ui.StatusLabel.Text = "状态: 目标未加载"
            return
        end
        -- 距离检查
        local myPart = getAimPart(LocalPlayer)
        if not myPart then
            ui.StatusLabel.Text = "状态: 本地角色未加载"
            return
        end
        local dist = (myPart.Position - aimPart.Position).Magnitude
        if settings.maxDistance and dist > settings.maxDistance then
            ui.StatusLabel.Text = string.format("状态: 目标过远（%.1f > %.1f）", dist, settings.maxDistance)
            return
        end
        -- 墙壁检测（本地与目标之间是否被挡）
        if settings.wallCheck then
            local clear = isLineOfSightClear(aimPart)
            if not clear then
                ui.StatusLabel.Text = "状态: 无法锁定 — 与目标之间有阻挡"
                return
            end
        end
        -- 允许锁定
        settings.locked = true
        ui.LockBtn.Text = "解锁"
        ui.LockBtn.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
        ui.Crosshair.Visible = true
        ui.StatusLabel.Text = "状态: 已锁定"
    else
        -- 解锁
        settings.locked = false
        ui.LockBtn.Text = "锁定"
        ui.LockBtn.BackgroundColor3 = Color3.fromRGB(50,120,210)
        ui.Crosshair.Visible = false
        ui.StatusLabel.Text = "状态: 已解锁"
    end
end

ui.LockBtn.MouseButton1Click:Connect(tryToggleLock)

-- 快捷键：P 切换锁定，U 切换 UI 显示，[ / ] 切换目标
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.UserInputType == Enum.UserInputType.Keyboard then
        local key = input.KeyCode
        if key == Enum.KeyCode.P then
            tryToggleLock()
        elseif key == Enum.KeyCode.U then
            ui.ScreenGui.Enabled = not ui.ScreenGui.Enabled
        elseif key == Enum.KeyCode.LeftBracket or key == Enum.KeyCode.Comma then
            selectTargetByIndex(settings.targetIndex - 1)
            refreshTargetLabel()
        elseif key == Enum.KeyCode.RightBracket or key == Enum.KeyCode.Period then
            selectTargetByIndex(settings.targetIndex + 1)
            refreshTargetLabel()
        end
    end
end)

-- 玩家离开时处理
Players.PlayerRemoving:Connect(function(p)
    if settings.target == p then
        settings.target = nil
        selectTargetByIndex(settings.targetIndex)
        refreshTargetLabel()
        if settings.locked then
            -- 自动解锁
            settings.locked = false
            ui.LockBtn.Text = "锁定"
            ui.LockBtn.BackgroundColor3 = Color3.fromRGB(50,120,210)
            ui.Crosshair.Visible = false
            ui.StatusLabel.Text = "状态: 目标离开，已解锁"
        end
    end
end)

-- 主循环：更新准心位置与自动检测（每帧）
local crossPos = Vector2.new(0, 0) -- 当前像素坐标
RunService.RenderStepped:Connect(function(dt)
    -- 同步 UI 显示数据
    ui.DistBox.Text = tostring(settings.maxDistance)
    ui.SmoothBox.Text = tostring(settings.smoothing)
    refreshTargetLabel()

    if settings.locked and settings.target then
        local aimPart = getAimPart(settings.target)
        local myPart = getAimPart(LocalPlayer)
        if not aimPart or not myPart then
            -- 角色未加载 -> 解锁
            settings.locked = false
            ui.LockBtn.Text = "锁定"
            ui.LockBtn.BackgroundColor3 = Color3.fromRGB(50,120,210)
            ui.Crosshair.Visible = false
            ui.StatusLabel.Text = "状态: 角色未加载，已解锁"
            return
        end

        -- 距离检测
        local dist = (myPart.Position - aimPart.Position).Magnitude
        if settings.maxDistance and dist > settings.maxDistance then
            settings.locked = false
            ui.LockBtn.Text = "锁定"
            ui.LockBtn.BackgroundColor3 = Color3.fromRGB(50,120,210)
            ui.Crosshair.Visible = false
            ui.StatusLabel.Text = string.format("状态: 目标过远（%.1f > %.1f），已解锁", dist, settings.maxDistance)
            return
        end

        -- 墙壁检测（本地玩家与目标之间）
        if settings.wallCheck then
            local clear = isLineOfSightClear(aimPart)
            if not clear then
                settings.locked = false
                ui.LockBtn.Text = "锁定"
                ui.LockBtn.BackgroundColor3 = Color3.fromRGB(50,120,210)
                ui.Crosshair.Visible = false
                ui.StatusLabel.Text = "状态: 与目标被墙隔开，已解锁"
                return
            end
        end

        -- 计算世界瞄准点（部位位置 + 偏移）
        local worldPoint = aimPart.Position + settings.aimOffset

        -- 将世界坐标转换为屏幕坐标
        local viewportPoint = Camera:WorldToViewportPoint(worldPoint)
        local vx, vy, vz = viewportPoint.X, viewportPoint.Y, viewportPoint.Z
        local onScreen = vz > 0 -- 在相机前面且可见

        if not onScreen then
            ui.StatusLabel.Text = "状态: 目标不在视野内"
            -- 仍然保持锁定但隐藏准心或可选择解锁；这里我们选择继续显示（不可见）并提示
        else
            ui.StatusLabel.Text = "状态: 锁定并跟随"
        end

        -- 目标像素坐标
        local targetPixel = Vector2.new(vx, vy)

        -- 初始化 crossPos 为当前位置（如果尚未设定）
        if crossPos.X == 0 and crossPos.Y == 0 then
            crossPos = targetPixel
        end

        -- 平滑移动到目标像素
        local alpha = math.clamp(dt * settings.smoothing, 0, 1)
        crossPos = crossPos:Lerp(targetPixel, alpha)

        -- 将像素坐标设置到 GUI（使用绝对像素位置）
        ui.Crosshair.Position = UDim2.new(0, crossPos.X, 0, crossPos.Y)

        -- 可选：当目标不在屏幕前方时隐藏准心
        ui.Crosshair.Visible = onScreen

    else
        -- 非锁定状态，确保准心隐藏
        if ui.Crosshair.Visible then
            ui.Crosshair.Visible = false
        end
        ui.StatusLabel.Text = "状态: 空闲"
    end
end)

-- 初始提示
if #getOtherPlayers() == 0 then
    ui.StatusLabel.Text = "状态: 等待其他玩家加入"
end

-- 在本地玩家重生时确保状态合理
LocalPlayer.CharacterAdded:Connect(function()
    if not settings.locked then
        ui.Crosshair.Visible = false
    end
end)

print("[AssistAim] 脚本已就绪。快捷键：P 切换锁定，U 隐藏/显示 UI，[/] 切换目标。")
