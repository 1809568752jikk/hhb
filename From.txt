-- 界面开关逻辑
ToggleButton.MouseButton1Click:Connect(function()
    MainFrame.Visible = not MainFrame.Visible
end)

-- [功能实现 - 保持逻辑不变]
FunctionButton.MouseButton1Click:Connect(function()
    print("XSimple: 正在执行绕过逻辑...")
    
    -- 你的原始逻辑开始
    game:GetService("Players").LocalPlayer:SetAttribute("mobileDealer",true)
    
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local mobileDealer = require(ReplicatedStorage.devv.shared.Indicies.mobileDealer)
    
    -- 修改现有物品库存
    for category, items in pairs(mobileDealer) do 
        for _, item in ipairs(items) do 
            item.stock = 999999 
        end 
    end 
    
    -- 插入特定物品
    table.insert(mobileDealer.Gun, {itemName = "Acid Gun", stock = 999999})
    table.insert(mobileDealer.Gun, {itemName = "Candy Bucket", stock = 999999})
    -- 你的原始逻辑结束
    
    FunctionButton.Text = "执行成功！"
    FunctionButton.BackgroundColor3 = Color3.fromRGB(40, 150, 40)
    wait(2)
    FunctionButton.Text = "绕过移动经销商系统"
    FunctionButton.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
end)

print("XSimple AI 助手提示：UI 已加载。点击 '隐藏/显示' 切换界面。")
