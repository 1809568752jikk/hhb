-- 滑块交互函数
local function sliderInput(sliderFrame, handle, minV, maxV, onChange)
	local dragging = false

	local function updateFromX(x)
		local absX = math.clamp(x - sliderFrame.AbsolutePosition.X, 0, sliderFrame.AbsoluteSize.X)
		local t = absX / sliderFrame.AbsoluteSize.X
		handle.Size = UDim2.new(t, 0, 1, 0)
		local value = minV + (maxV - minV) * t
		onChange(value)
	end

	sliderFrame.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			updateFromX(input.Position.X)
		end
	end)
	sliderFrame.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = false
		end
	end)
	sliderFrame.InputChanged:Connect(function(input)
		if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
			updateFromX(input.Position.X)
		end
	end)
	-- 支持点击滑道任意位置
	sliderFrame.MouseButton1Click = sliderFrame.MouseButton1Click or nil
end

-- 填充玩家列表
local function refreshPlayerList()
	-- 清空旧按钮
	for _, child in ipairs(uiList:GetChildren()) do
		if child:IsA("TextButton") then child:Destroy() end
	end

	for _, pl in ipairs(Players:GetPlayers()) do
		if pl ~= LocalPlayer then
			local btn = Instance.new("TextButton")
			btn.Size = UDim2.new(1, -8, 0, 26)
			btn.BackgroundColor3 = Color3.fromRGB(48,48,48)
			btn.TextColor3 = Color3.fromRGB(230,230,230)
			btn.Font = Enum.Font.SourceSans
			btn.TextSize = 14
			btn.AutoButtonColor = true
			btn.Text = pl.Name
			btn.Parent = uiList

			btn.MouseButton1Click:Connect(function()
				targetPlayer = pl
				-- 高亮按钮显示当前目标
				for _, b in ipairs(uiList:GetChildren()) do
					if b:IsA("TextButton") then
						b.BackgroundColor3 = Color3.fromRGB(48,48,48)
					end
				end
				btn.BackgroundColor3 = Color3.fromRGB(80,120,180)
			end)
		end
	end
end

Players.PlayerAdded:Connect(function() wait(0.2); refreshPlayerList() end)
Players.PlayerRemoving:Connect(function(pl)
	if pl == targetPlayer then
		targetPlayer = nil
		isLocked = false
		lockToggle.Text = "Lock: OFF"
	end
	wait(0.2); refreshPlayerList()
end)

-- 锁定开关
lockToggle.MouseButton1Click:Connect(function()
	isLocked = not isLocked
	lockToggle.Text = "Lock: " .. (isLocked and "ON" or "OFF")
	-- 切换 CameraType
	local cam = Workspace.CurrentCamera
	if isLocked then
		-- 如果没有目标，试着自动选最近的玩家
		if not targetPlayer then
			-- 选择与鼠标或摄像机最近的玩家
			local closest, dmin = nil, math.huge
			for _, pl in ipairs(Players:GetPlayers()) do
				if pl ~= LocalPlayer and pl.Character and pl.Character:FindFirstChild("HumanoidRootPart") then
					local pos = pl.Character.HumanoidRootPart.Position
					local dist = (Workspace.CurrentCamera.CFrame.Position - pos).Magnitude
					if dist < dmin then
						dmin = dist; closest = pl
					end
				end
			end
			if closest then
				targetPlayer = closest
				refreshPlayerList()
			end
		end
		cam.CameraType = Enum.CameraType.Scriptable
	else
		Workspace.CurrentCamera.CameraType = Enum.CameraType.Custom
	end
end)

-- 射线检测函数（从目标到期望相机位置）
local function resolveCameraPosition(targetPos, desiredPos, ignoreList)
	ignoreList = ignoreList or {}
	local direction = desiredPos - targetPos
	local dist = direction.Magnitude
	if dist <= 0.001 then
		return desiredPos
	end
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Blacklist
	params.FilterDescendantsInstances = ignoreList
	params.IgnoreWater = true

	local result = Workspace:Raycast(targetPos, direction.Unit * dist, params)
	if result then
		-- 返回碰撞点前一个小偏移，避免穿模
		local safePos = result.Position + result.Normal * 0.5
		return safePos, true
	end
	return desiredPos, false
end

-- 主循环：在 RenderStepped 上更新摄像机
local camera = Workspace.CurrentCamera
local lastTime = tick()
RunService.RenderStepped:Connect(function(dt)
	camera = Workspace.CurrentCamera
	if isLocked and targetPlayer and targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") then
		local hrp = targetPlayer.Character.HumanoidRootPart
		local targetPos = hrp.Position + Vector3.new(0, CAMERA_HEIGHT, 0)

		-- 预测：使用目标速度估算未来位置
		local vel = Vector3.new(0,0,0)
		if hrp:FindFirstChild("Velocity") then
			vel = hrp.Velocity
		else
			vel = hrp.Velocity -- HumanoidRootPart 自带 Velocity
		end
		local predicted = targetPos + vel * predictFactor

		-- 期望摄像机位置：沿相对于目标的方向取反向（从目标背后看）
		-- 我们用当前摄像机与目标连线方向保持水平朝向：从目标往摄像机方向 normalized * distance
		local camPos = camera.CFrame.Position
		local backDir = (camPos - targetPos)
		if backDir.Magnitude < 0.01 then
			backDir = Vector3.new(0, 0, 1)
		end
		backDir = backDir.Unit

		local desiredPos = predicted + backDir * distance

		-- 墙体检测：从目标（或预测位置）朝 desiredPos 射线，黑名单过滤目标角色自身和玩家摄像机等
		local ignoreList = {LocalPlayer.Character, targetPlayer.Character}
		local resolvedPos, obstructed = resolveCameraPosition(predicted, desiredPos, ignoreList)

		-- 如果被阻挡，尝试从 targetPos 往摄像机方向做第二次射线（更保守）
		if obstructed then
			-- resolvedPos 已经是碰撞点前的安全点
		end

		-- 让摄像机朝向目标（使用预测目标位置）
		local lookAt = predicted

		-- 平滑插值摄像机 CFrame
		local desiredCFrame = CFrame.new(resolvedPos, lookAt)
		camera.CFrame = camera.CFrame:Lerp(desiredCFrame, math.clamp(SMOOTHNESS * dt, 0, 1))

		-- 可选：如果目标死亡或不可见，则取消锁定（防止卡住）
		local humanoid = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
		if not humanoid or humanoid.Health <= 0 then
			isLocked = false
			lockToggle.Text = "Lock: OFF"
			camera.CameraType = Enum.CameraType.Custom
			targetPlayer = nil
		end
	else
		-- 当未锁定时确保相机模式为 Custom（让玩家自由控制）
		if camera.CameraType == Enum.CameraType.Scriptable and not isLocked then
			camera.CameraType = Enum.CameraType.Custom
		end
	end
end)

-- 初始设置（设置滑块可见位置）

	-- 设置初始 handle 大小

	local tp = (DEFAULT_PREDICT - MIN_PREDICT) / (MAX_PREDICT - MIN_PREDICT)
	predictHandle.Size = UDim2.new(tp, 0, 1, 0)
end

-- 额外热键（按键 'L' 切换锁定）
UserInputService.InputBegan:Connect(function(input, processed)
	if processed then return end
	if input.KeyCode == Enum.KeyCode.L then
		lockToggle:CaptureFocus() -- 防止文字框捕获？
		lockToggle.MouseButton1Click:Connect(function() end) -- noop to ensure consistent state
		isLocked = not isLocked
		lockToggle.Text = "Lock: " .. (isLocked and "ON" or "OFF")
		if isLocked then
			camera.CameraType = Enum.CameraType.Scriptable
		else
			camera.CameraType = Enum.CameraType.Custom
		end
	end
end)
